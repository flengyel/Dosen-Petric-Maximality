def code(s):
 # Preserve strings and erase nested block and line comments, then compare tokens.
 out=[]; i=0
 while i<len(s):
  if s.startswith('/-',i):
   depth=1; i+=2
   while depth:
    assert i<len(s), 'unterminated comment'
    if s.startswith('/-',i): depth+=1; i+=2
    elif s.startswith('-/',i): depth-=1; i+=2
    else: i+=1
   out.append(' ')
  elif s.startswith('--',i):
   end=s.find('\n',i); i=len(s) if end<0 else end
   out.append(' ')
  elif s[i]=='"':
   start=i; i+=1
   while i<len(s):
    if s[i]=='\\': i+=2
    elif s[i]=='"': i+=1; break
    else: i+=1
   out.append(s[start:i])
  else: out.append(s[i]); i+=1
 return '\n'.join(''.join(out).split())
